extends SceneTree
## Controls' knock audit (scratch probe): one scripted pad on the real input hub plays seeded matches against the computer, and every pad knock is read against the cue that caused it.
##   godot --headless --path <tree> --script res://tools/feel/knock_audit.gd -- <matches> <baseSeed> --arm=stick|chaos --level=easy|medium|hard [--wall=150]
## stick: right-stick flicks every 8 to 24 ticks, nothing else. chaos: the same flicks plus X, Y or B every 5 to 9 ticks (the full queue, the chord, the stun string).
## Wired as the host wires it (render/core/main.gd _haptics_from_events): every cue through SimHaptics.requests, every cue of stick_start / stick_refused / press_refused answers the hub's wait.
const QaPlanet = preload("res://qa/godot/qa_planet.gd")

var hub: SimInputHub
var held: Dictionary = {}


func _init() -> void:
	var pos: Array = []
	var arm: String = "stick"
	var level: String = "easy"
	var wall: float = 150.0
	for a in OS.get_cmdline_user_args():
		if a.begins_with("--arm="):
			arm = a.substr(6)
		elif a.begins_with("--level="):
			level = a.substr(8)
		elif a.begins_with("--wall="):
			wall = float(a.substr(7))
		else:
			pos.append(a)
	var n: int = int(pos[0]) if pos.size() > 0 else 3
	var base: int = int(pos[1]) if pos.size() > 1 else 1
	SimInputData.load_and_apply()
	var dai = load("res://sim/director/ai.gd")
	dai.set("level", level)
	var acc := {"cuesByKey": {}, "knocksByReason": {}, "hubKnocksByKind": {}, "hubKnocksByReason": {}, "knocksSameTickAsLanded": 0, "landed": 0, "refusalCues": 0, "knocks": 0, "presses": 0, "stickStart": 0, "stickHit": 0, "silentRefusals": {}, "unknownReasons": {}, "perMatch": []}
	for m in range(n):
		var seed_: int = base + m
		var S := _make(seed_)
		if QaPlanet.fellBack(S, seed_):
			quit(3)
			return
		var rng := RandomNumberGenerator.new()
		rng.seed = seed_
		var st := {"next": 20, "rel": {}, "rsTo": -1, "aim": Vector2.ZERO, "presses": 0, "bnext": 30}
		_btn("west", true)
		st.rel[2] = ["west"]
		var t0: int = Time.get_ticks_msec()
		var koSeen: int = -1
		for t in range(int(900.0 * 60.0)):
			_drive(arm, S, t, st, rng)
			_step(S, acc)
			if S.game.ko != null and koSeen < 0:
				koSeen = t
			if koSeen >= 0 and S.game.koT > 3.0:
				break
			if (Time.get_ticks_msec() - t0) / 1000.0 > wall:
				break
		acc.presses += int(st.presses)
		for k in held.keys():
			_btn(k, false)
		SimCore.dispose(S)
	acc["arm"] = arm
	acc["level"] = level
	acc["n"] = n
	print("KNOCKAUDIT " + JSON.stringify(acc))
	quit()


func _make(seed_: int) -> SimState:
	hub = SimInputHub.new()
	held = {}
	var su: Dictionary = QaPlanet.setup(seed_, hub.setup())
	su["intro"] = false
	var S := SimCore.createSim()
	SimCore.newMatch(S, seed_, {"p1": false, "p2": true}, su)
	hub.set_pad_has_right_stick(0, true)
	return S


func _btn(name: String, down: bool) -> void:
	if down == held.has(name):
		return
	if down:
		held[name] = true
	else:
		held.erase(name)
	hub.pad_button(0, name, down)


func _walk(S: SimState) -> void:
	var dx: float = SimWrap.sdx(S.fighters[0].x, S.fighters[1].x)
	var dy: float = S.fighters[1].y - S.fighters[0].y
	hub.pad_stick(0, signf(dx) if absf(dx) > 110.0 else 0.0, signf(dy) if absf(dy) > 40.0 else 0.0)


func _aim(S: SimState, dir: int) -> Vector2:
	var side: float = 1.0 if SimWrap.sdx(S.fighters[0].x, S.fighters[1].x) >= 0.0 else -1.0
	match dir:
		1:
			return Vector2(0.0, 1.0)
		2:
			return Vector2(0.0, -1.0)
		3:
			return Vector2(side, 0.0)
	return Vector2(-side, 0.0)


func _tap(st: Dictionary, t: int, names: Array) -> void:
	for nm in names:
		_btn(nm, true)
	st.rel[t + 2] = names
	st.presses = int(st.presses) + 1


func _drive(arm: String, S: SimState, t: int, st: Dictionary, rng: RandomNumberGenerator) -> void:
	if st.rel.has(t):
		for nm in st.rel[t]:
			_btn(nm, false)
		st.rel.erase(t)
	if t >= int(st.next):
		st.aim = _aim(S, rng.randi_range(1, 4))
		st.rsTo = t + 3
		st.presses = int(st.presses) + 1
		st.next = t + rng.randi_range(8, 24)
	if t < int(st.rsTo):
		hub.pad_stick_right(0, st.aim.x, st.aim.y)
	elif t == int(st.rsTo):
		hub.pad_stick_right(0, 0.0, 0.0)
	if arm == "chaos" and t >= int(st.bnext):
		var r: float = rng.randf()
		_tap(st, t, ["west" if r < 0.5 else ("north" if r < 0.8 else "east")])
		st.bnext = t + rng.randi_range(5, 9)
	_walk(S)


func _bump(d: Dictionary, k: String) -> void:
	d[k] = int(d.get(k, 0)) + 1


func _step(S: SimState, acc: Dictionary) -> void:
	hub.set_humans(S.fighters[0].ai == null, S.fighters[1].ai == null)
	hub.set_struggle(0, SimDir.inStruggle(S, S.fighters[0]), DirBeam.pushModeOf(S, S.fighters[0]) == "stick")
	var it: SimIntent = hub.intent(0)
	var pausing: bool = S.pause.left > 0
	if SimCore.step(S, [it, null]):
		hub.consumed()
	elif pausing:
		if S.pause.left > 0:
			hub.drop_edges()
		else:
			hub.resume()
	var landed_now: bool = false
	var knocks_now: int = 0
	for e in S.out.fx:
		var typ: String = str(e.type)
		if typ == "damage" and int(e.attacker) == 0 and int(e.victim) != 0:
			landed_now = true
			acc.landed += 1
		if typ != "cue":
			continue
		var who: int = int(e.actor)
		var k: String = str(e.kind)
		if who == 0:
			if k == "stick_start":
				acc.stickStart += 1
			elif k == "stick_hit":
				acc.stickHit += 1
			if k == "stick_start" or k == "stick_refused" or k == "press_refused":
				hub.note_press_answered(0)
		if who == 0 and (k == "stick_refused" or k == "press_refused"):
			acc.refusalCues += 1
			_bump(acc.cuesByKey, k + "/" + str(e.source))
		for r in SimHaptics.requests(k, who, int(e.target), str(e.text), float(e.amount), -1, str(e.source)):
			if str(r.get("kind", "")) == "stick_refused" and int(r.get("slot", -1)) == 0:
				acc.knocks += 1
				knocks_now += 1
				var why: String = str(r.get("reason", "?"))
				_bump(acc.knocksByReason, why)
				if why == "other" or why == "busy" or why == "stun_limit" or why == "replaced":
					_bump(acc.unknownReasons, why + " <- " + k + "/" + str(e.source) + "/" + str(e.text))
		if who == 0 and (k == "stick_refused" or k == "press_refused"):
			if SimHaptics.requests(k, who, int(e.target), str(e.text), float(e.amount), -1, str(e.source)).is_empty():
				_bump(acc.silentRefusals, k + "/" + str(e.source))
	if landed_now and knocks_now > 0:
		acc.knocksSameTickAsLanded += 1
	for h in hub.take_haptics():
		if str(h.get("kind", "")) == "stick_refused":
			_bump(acc.hubKnocksByKind, "stick_refused")
			_bump(acc.hubKnocksByReason, str(h.get("reason", "?")))
	hub.take_refusals()
	hub.on_events(S.out.fx)
	S.out.fx.clear()
	S.out.feed.clear()
