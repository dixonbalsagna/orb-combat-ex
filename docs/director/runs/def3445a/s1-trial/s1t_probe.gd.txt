extends SceneTree
## SPIKE S1 trial's escape probe (scratch only): QA's X-16 isn't built, so this is the trial's own, after the stand-in's design section 15.
## A scripted shooter (a full fighter: six stages) whose stick tracks the defender to the nearest of 32 directions fires stages 1, 3 and 6
## from three ranges inside each stage's reach (stage 1: 1,500, 3,000, 4,000; stages 3 and 6: 1,500, 3,000, 6,000), both 600 above the
## ground. Four scripted defenders: stands; has been flying straight up (across a level line) at plain speed; stands, then dashes up 15
## ticks (0.25 s) after the fire; has been dashing up. Per beam: the outcome (HIT, GUARD, DODGE, PASS, DEFLECT, MISS), ticks from the fire
## to contact, how far off the line the defender was as the front passed, ticks from the release to the fire, and whether the defender
## was ever held before the beam reached him.
##   godot --headless --path . --script res://sim/director/tools/s1t_probe.gd [-- --off]   (--off: today's beam, for comparison)

var _ev: Array = []


func _init() -> void:
	var off: bool = OS.get_cmdline_user_args().has("--off")
	DirStick.setForce(1)
	DirBeamPlay.setFrontForce(1)
	DirBeamMove.setForce(1)
	DirBeamAim.setForce(1)
	DirBeamSpace.setForce(0 if off else 1)
	var rows: Array = []
	for stage in [1, 3, 6]:
		for gap in ([1500.0, 3000.0, 4000.0] if stage == 1 else [1500.0, 3000.0, 6000.0]):
			for how in ["stands", "flying", "dash15", "dashing"]:
				var r: Dictionary = _case(stage, gap, how)
				rows.append(r)
				print("S1T %s stage %d (fired %d) gap %4d %-7s: %-7s contact +%s ticks, off the line %s (hits within %d), release to fire %s, held before %s" % ["off" if off else "on ", stage, r.st, int(gap), how, r.out, str(r.contact), str(r.off), int(DirBeamSpace.half(stage) + 40.0), str(r.delay), str(r.held)])
	print("== X-16 rows (hit = HIT or GUARD; a moving target is flying, dash15 or dashing)")
	for stage in [1, 3, 6]:
		var line := PackedStringArray()
		for gap in ([1500.0, 3000.0, 4000.0] if stage == 1 else [1500.0, 3000.0, 6000.0]):
			var cell := PackedStringArray()
			for r in rows:
				if r.stage == stage and r.gap == gap:
					cell.append(r.how.substr(0, 4) + " " + ("hit" if (r.out == "HIT" or r.out == "GUARD") else r.out.to_lower()))
			line.append("%d: %s" % [int(gap), ", ".join(cell)])
		print("S1T stage %d | %s" % [stage, " | ".join(line)])
	var mv_far: int = 0
	var mv_far_n: int = 0
	var mv_near: int = 0
	var mv_near_n: int = 0
	var bad: int = 0
	for r in rows:
		if r.how != "stands":
			if r.gap > 3000.0:
				mv_far_n += 1
				mv_far += 1 if (r.out == "HIT" or r.out == "GUARD") else 0
			if r.gap <= 1500.0:
				mv_near_n += 1
				mv_near += 1 if (r.out == "HIT" or r.out == "GUARD") else 0
		if r.held or (r.delay > 6 and r.delay >= 0):
			bad += 1
	print("S1T bands: moving target beyond 3,000 hit %d of %d (band 10 to 30%%); moving inside 1,500 hit %d of %d (band 40 to 70%%); beams late or a target held before contact: %d" % [mv_far, mv_far_n, mv_near, mv_near_n, bad])
	quit(0)


func _step(S: SimState, a: SimIntent, b: SimIntent) -> void:
	SimCore.step(S, [a, b])
	for e in S.out.fx:
		if e.type == "cue":
			_ev.append([S.tick, "cue", String(e.kind), String(e.text), String(e.source), int(e.actor)])
		elif e.type == "beam_outcome":
			_ev.append([S.tick, "outcome", String(e.kind), "", "", int(e.actor)])
	S.out.fx.clear()
	S.out.feed.clear()


func _case(stage: int, gap: float, how: String) -> Dictionary:
	var S := SimCore.createSim()
	SimCore.newMatch(S, 11, {"p1": false, "p2": false}, {"intro": false})
	var A = S.fighters[0]
	var D = S.fighters[1]
	for f in S.fighters:
		f.ai = null
		f.act.v2 = true
		f.hp = 100000.0
		f.ki = 100.0
	A.tier = 4.0
	if "power" in A:
		A.power = 100.0
	for _k in range(5):
		_step(S, SimIntent.new(), SimIntent.new())
	A.y = WorldTerrain.groundY(S, A.x, A.z) + 600.0
	D.x = SimWrap.wrap(A.x + gap)
	D.y = A.y
	A.vx = 0.0
	A.vy = 0.0
	D.vx = 0.0
	D.vy = 0.0
	for _k in range(40):   # settle in the air, and D free long enough to act
		_step(S, SimIntent.new(), _def(how, -1, false))
		A.y = WorldTerrain.groundY(S, A.x, A.z) + 600.0
	_ev.clear()
	var holdT: int = int(DirBeamMove.cfg().stageTicks[stage - 1]) + 2
	var out := {"stage": stage, "gap": gap, "how": how, "out": "", "contact": -1, "off": -1, "delay": -1, "held": false, "st": -1}
	var relT: int = -1
	var fireT: int = -1
	for t in range(holdT + 260):
		var a := SimIntent.new()
		a.rsDir = SimStrikeRead.RIGHT
		a.rsMask = DirStick.RB
		a.stanceMask = DirStick.RB
		a.rsAim = DirBeamSpace.sectorTo(S, A, D)
		if t == 0:
			a.rsStart = true
			a.rsHeld = true
		elif t < holdT:
			a.rsHeld = true
		elif t == holdT:
			a.rsRelease = true
			relT = S.tick + 1
		else:
			a = SimIntent.new()
		var since: int = (S.tick - fireT) if fireT >= 0 else -1
		_step(S, a, _def(how, since, fireT >= 0))
		if fireT < 0:
			for e in _ev:
				if e[1] == "cue" and e[2] == "beam_fire" and e[5] == 0:
					fireT = e[0]
					out.delay = fireT - relT
					break
		elif out.out == "" and D.state == "locked":
			out.held = true
		if fireT >= 0 and out.out == "":
			for e in _ev:
				if e[1] == "outcome" and e[5] == 0 and e[0] >= fireT:
					out.out = String(e[2])
					out.contact = e[0] - fireT
					break
		if out.out != "":
			break
	for e in _ev:
		if e[1] == "cue" and e[2] == "beam_miss":
			out.off = int(e[3])
		if e[1] == "cue" and e[2] == "beam_stage" and e[5] == 0:
			out.st = int(e[3])
	if out.out == "":
		out.out = "none"
	SimCore.dispose(S)
	return out


## The defender's stick: since is the ticks since the fire (-1 before it).
func _def(how: String, since: int, fired: bool) -> SimIntent:
	var i := SimIntent.new()
	match how:
		"flying":
			i.my = 1.0
		"dashing":
			i.my = 1.0
			i.dash = true
			i.sprint = true
		"dash15":
			if fired and since >= 15:
				i.my = 1.0
				i.dash = true
				i.sprint = true
	return i
