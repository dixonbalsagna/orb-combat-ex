#!/usr/bin/env python3
"""Game Design's ruling on Simulation's hitstop-any (gauntlet-plan.md s7, 592a807b), in a scratch tree (NOT a park):
hitstop.json budgetAny on, budget.share 0.12, cutMul 0.7, heavy joins budgetAny.neverCut; and so 'heavy' reaches the brawl's blows,
SimHitstop.cutPassed takes the blow's kind (ex.kind) when the hit names no weight (the park reads only o.weight, which brawl blows
don't carry). python hs_ruled.py <tree>"""
import sys, os, re
root = sys.argv[1]
def edit(rel, pairs, regex=False):
    p = os.path.join(root, rel)
    s = open(p, encoding='utf8', newline='').read()
    for a, b in pairs:
        if regex:
            s, n = re.subn(a, b, s); assert n == 1, (rel, a)
        else:
            assert s.count(a) == 1, (rel, a[:80]); s = s.replace(a, b)
    open(p, 'w', encoding='utf8', newline='').write(s)
edit('data/fight/hitstop.json', [
    (r'("budget": \{\s*"share": )[0-9.]+', r'\g<1>0.12'),
    (r'"cutMul": [0-9.]+', '"cutMul": 0.7'),
    (r'"budgetAny": \{"enabled": false, "cutFloor": 1, "neverCut": \["decisive", "ko", "launch", "launch_full"\]\}',
     '"budgetAny": {"enabled": true, "cutFloor": 1, "neverCut": ["decisive", "ko", "launch", "launch_full", "heavy"]}'),
], regex=True)
edit('sim/core/hitstop.gd', [
    ('static func cutPassed(S: SimState, o, passed: float) -> float:\n',
     'static func cutPassed(S: SimState, o, passed: float, kind: String = "") -> float:\n'),
    ('\tvar w = o.get("weight") if o is Dictionary else null\n\tif w is String and anyNever.has(w):\n\t\treturn passed\n\tvar n: int = int(SimMathx.jround(passed / SimConst.DT))',
     '\tvar w = o.get("weight") if o is Dictionary else null\n\tif not (w is String) and kind != "":\n\t\tw = kind   # SCRATCH: the blow\'s kind when the hit names no weight (brawl blows)\n\tif w is String and anyNever.has(w):\n\t\treturn passed\n\tvar n: int = int(SimMathx.jround(passed / SimConst.DT))'),
])
if 'nohp' in sys.argv[2:]:
    # SCRATCH: HP is a readout floored at 0 since S2 (HP_ENDS_MATCH false), so 'hp <= 0' isn't the KO: once his bar is empty every blow
    # on him would keep its whole stop. The KO's blow only.
    edit('sim/core/hitstop.gd', [
        ('\tvar keep: bool = S.game.ko == D or D.hp <= 0.0',
         '\tvar keep: bool = S.game.ko == D   # SCRATCH: hp is a readout (HP_ENDS_MATCH false), not the KO'),
    ])
edit('sim/core/damage.gd', [
    ('(SimHitstop.cutPassed(S, o, jor(o.get("stop", 0.0), 0.05)) if S.hsAny',
     '(SimHitstop.cutPassed(S, o, jor(o.get("stop", 0.0), 0.05), String(ex.kind) if ex != null else "") if S.hsAny'),
])
print('ruled applied: share 0.12, cutMul 0.7, neverCut + heavy, kind reaches cutPassed')
