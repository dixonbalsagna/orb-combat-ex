## The calm turn's read on a gen 2 planet (calm-turn-read.md; Game Design's rulings-2026-10-26 item 28): a form step's
## ready tell begins the warm, with World's smallest bearing as the fallback, at K columns a tick in all
## (calmTurn.warmColsGen2; World's turn.warmTotal: each ring reads its share); on a gen 1 planet the ready tell asks
## nothing, as before; with the director's calm turn off nothing runs on either.
func _calmGen2() -> String:
	var ct: Dictionary = DirInterrupt.data().calmTurn
	var was: bool = bool(ct.enabled)
	var quiet: bool = FighterData.quiet
	FighterData.quiet = true
	var K: int = DirCalm.gen2Cols()
	var got: Array = []
	for g in [1, 2]:
		var S := SimCore.createSim()
		ct.enabled = false
		SimCore.newMatch(S, 3, {}, {"planet": 7, "planetGen": g, "touch": 3, "arena": {"cols": 2400, "turn": true}, "intro": false})
		if S.epoch == null:
			ct.enabled = was
			FighterData.quiet = quiet
			return "gen %d: no calm turn on a planet match with arena.turn" % g
		S.epoch.lastT = S.T - 61.0   # past World's 60 s between turns (a test's shortcut)
		SimFx.transformReady(S, S.fighters[0], 2.0, "test")
		var off: bool = S.epoch.next == null
		ct.enabled = true
		SimFx.transformReady(S, S.fighters[0], 2.0, "test")
		var nx = S.epoch.next
		got.append({"gen": g, "off": off, "asked": nx != null, "why": String(nx.why) if nx != null else "", "k": int(nx.k) if nx != null else 0,
			"rings": S.epoch.warm.size(), "rate": DirCalm.warmRate(S) if nx != null else 0})
		SimCore.dispose(S)
	ct.enabled = was
	WorldPlanet.forget()
	WorldClimate.useGen(0)
	FighterData.quiet = quiet
	var g1: Dictionary = got[0]
	var g2: Dictionary = got[1]
	if not g1.off or g1.asked or not g2.off or not g2.asked or g2.why != "ready" or int(g2.k) != K or int(g2.rings) < 1 or int(g2.rings) > 2 or int(g2.rate) != K / int(g2.rings):
		return "the calm turn on gen 2: %s (want gen 1's ready tell asking nothing; gen 2's asking 'ready' at %d columns a tick in all, its rings sharing them; none with the turn off)" % [str(got), K]
	return ""
