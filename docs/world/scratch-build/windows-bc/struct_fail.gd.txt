# ===================================================================== C: the failing state (docs/world/destruction-state.md section 3)

## Is x, z (a fighter's place) under the stack of building b while it falls? A pancake drops straight down over the footprint and 1 bh beside it; a tip
## also throws the stack out on its side (failTipReach of the height above the break, at most failTipMaxBh). In depth: within failReachZ of the footprint.
static func _inFailReach(S: SimState, b, f) -> bool:
	var c: Dictionary = WorldBrunt.cfg()
	var ex: float = SimWrap.sdx(b.x, f.x)
	var half: float = b.w * 0.5 + float(c.failBesideBh) * 75.0
	var lo: float = -half
	var hi: float = half
	if b.failDir != 0:
		var reach: float = minf(float(c.failTipReach) * curH(b), float(c.failTipMaxBh) * 75.0)
		if b.failDir > 0:
			hi = b.w * 0.5 + reach
		else:
			lo = -(b.w * 0.5 + reach)
	if ex < lo or ex > hi:
		return false
	if absf(f.z - b.z) > b.d * 0.5 + float(c.failReachZ) * 75.0:
		return false
	return f.y <= baseY(S, b) + curH(b) + WorldBrunt.FLOOR_H


## A tower begins to come down: a warning (failWarn seconds, and over any hold a fighter under it is in by failWarnOver), then a floor lands each step
## from from_floor up. cx is where the blow fell (see failTipAt, failTipFar for which way it tips, if it does).
static func startFail(S: SimState, b, from_floor: int, cause, cx: float) -> void:
	if b.failK >= 0 or not b.alive:
		return
	var c: Dictionary = WorldBrunt.cfg()
	var warn: float = float(c.failWarn)
	var hurts: String = str(c.failHurts)
	b.failK = from_floor
	var u: float = SimWrap.sdx(b.x, cx) / (b.w * 0.5)
	if absf(u) > 1.0:   # a blow from beside it pushes it over, away from the blow (a far one, past failTipFar half widths, leaves it to fall in place)
		b.failDir = 0 if absf(u) > float(c.failTipFar) else (-1 if u > 0.0 else 1)
	else:   # a hit in the footprint: it tips toward the side of the breach once that is failTipAt of the half width off the middle
		b.failDir = 0 if absf(u) < float(c.failTipAt) else (1 if u > 0.0 else -1)
	for f in S.fighters:
		if _inFailReach(S, b, f):
			var hold: float = maxf(maxf(float(f.stunTicks), float(f.embedT)) / 60.0, maxf(S.dirS.stop, 0.0))
			warn = maxf(warn, hold + float(c.failWarnOver))
	b.failT = -warn
	b.failN = 0
	b.failBy = int(WorldCrater._slot(S, cause)) if cause != null else -1
	b.failHurt = PackedFloat32Array()
	b.failHurt.resize(2)
	b.failHurt.fill(0.0)
	var m: int = 0
	for k in range(from_floor, b.floors):
		if WorldBrunt.isStanding(b, k):
			m += 1
	var stepDur: float = maxf(float(c.failStep), float(c.failMin) / float(maxi(m, 1)))
	SimFx.buildingFailing(S, b, baseY(S, b), curH(b), warn, stepDur * float(m), b.failDir, float(b.failBy))
	for f in S.fighters:
		if _inFailReach(S, b, f):
			SimFx.hazardTelegraph(S, f, "collapse", warn, b.x)


## One tick of every tower that is coming down.
static func stepFailing(S: SimState, dt: float) -> void:
	for b in S.buildings:
		if b.failK >= 0:
			_failStep(S, b, dt)


static func _failStep(S: SimState, b, dt: float) -> void:
	var c: Dictionary = WorldBrunt.cfg()
	b.failT += dt
	if b.failT < 0.0:
		return   # the warning
	var m: int = 0
	for k in range(b.failK, b.floors):
		if WorldBrunt.isStanding(b, k):
			m += 1
	var stepDur: float = maxf(float(c.failStep), float(c.failMin) / float(maxi(m + b.failN, 1)))
	var due: int = int(floor(b.failT / stepDur))
	var cause = S.fighters[b.failBy] if b.failBy >= 0 and b.failBy < S.fighters.size() else null
	while b.failN < due and b.failK < b.floors:
		var k: int = b.failK
		while k < b.floors and not WorldBrunt.isStanding(b, k):
			k += 1
		if k >= b.floors:
			b.failK = b.floors
			break
		var s0: int = stage(b)
		var popF: float = WorldBrunt.floorPop(b)
		b.fmask = b.fmask & ~(1 << k)
		b.failK = k + 1
		b.failN += 1
		if popF > 0.0:
			WorldCollateral.kill(S, b.idx, popF, cause, 0.0, b.x, k)
		SimFx.floorsFall(S, b, k, k, 1)
		stageEmit(S, b, s0, cause, b.x, 1)
		_failHurt(S, b, k, cause)
	# the stack is down
	var left: bool = false
	for k2 in range(b.failK, b.floors):
		if WorldBrunt.isStanding(b, k2):
			left = true
			break
	if not left:
		b.failK = -1
		b.failT = 0.0
		b.failDir = 0
		if WorldBrunt.standingCount(b) == 0 and b.alive:
			b.hp = minf(b.hp, 0.0)
			damageBuilding(S, b, 0.001, cause, "implode", b.x, 0.0, true)   # the whole building is down: it falls, the heap and the people as always


## A floor lands: a fighter under it takes failDamageFloor, up to failDamageCap from this building, and is buried failBuryTicks. Whom: failHurts.
static func _failHurt(S: SimState, b, k: int, cause) -> void:
	var c: Dictionary = WorldBrunt.cfg()
	var hurts: String = str(c.failHurts)
	if hurts == "none":
		return
	for slot in range(S.fighters.size()):
		var f = S.fighters[slot]
		if hurts == "rival" and slot == b.failBy:
			continue
		if not _inFailReach(S, b, f) or b.failHurt.size() < 2:
			continue
		var room: float = float(c.failDamageCap) - b.failHurt[slot]
		if room <= 0.0:
			continue
		var d: float = minf(float(c.failDamageFloor), room)
		b.failHurt[slot] += d
		SimDamage.hurt(S, f, d, cause if cause != null else SimRoster.opp(S, f))
		f.stunTicks = maxi(f.stunTicks, int(c.failBuryTicks))
