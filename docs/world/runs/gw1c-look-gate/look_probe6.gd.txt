extends SceneTree
# A weaker crater first, then a stronger blow near it: does the stronger blow dig, and how much (the centre of the second blow, before and after).
const FIELD: Dictionary = {"intro": false, "planet": 7, "touch": 1, "planetGen": 3}
const E_BASE: float = (356.0 / 160.0) * (356.0 / 160.0)
func _mk() -> SimState:
	WorldClimate.useGen(3)
	WorldPlanet.forget()
	var S := SimCore.createSim()
	SimCore.newMatch(S, 77, {"p1": false, "p2": false}, FIELD)
	for t in range(5):
		SimCore.step(S)
	return S
func _init() -> void:
	print("E. a first blow of x1 energy, then a stronger blow `off` units away: depth the second adds at its own centre")
	for e2 in [2.0, 4.0, 9.0]:
		var row: String = ""
		for off in [0.0, 60.0, 120.0, 200.0, 300.0, 500.0]:
			var S := _mk()
			var x: float = S.fighters[0].x + 900.0
			WorldCrater.dig(S, x, E_BASE, S.fighters[0], "impact")
			var h0: float = WorldTerrain.groundY(S, x + off)
			var r = WorldCrater.dig(S, x + off, E_BASE * e2, S.fighters[0], "impact")
			row += "  off%d:%s%d" % [int(off), "" if r != null else "REFUSED ", int(h0 - WorldTerrain.groundY(S, x + off))]
		print("ESTR x1 first, then x%d:%s" % [int(e2), row])
	var S0 := _mk()
	var xa: float = S0.fighters[0].x + 900.0
	var alone: String = ""
	for e2 in [2.0, 4.0, 9.0]:
		var A := _mk()
		var h0: float = WorldTerrain.groundY(A, xa)
		WorldCrater.dig(A, xa, E_BASE * e2, A.fighters[0], "impact")
		alone += "  x%d:%d" % [int(e2), int(h0 - WorldTerrain.groundY(A, xa))]
	print("ESTR for scale, the same blow on fresh ground:%s" % alone)
	quit(0)
