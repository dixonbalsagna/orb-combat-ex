extends SceneTree
# Can a pure detail term restore what a 512-unit ground loses? The ring's exact column height (farColumn = surfAt + the surface detail) against variants built from nodes every 16th column (512 units):
#   V0  the line between the nodes' full heights (what a 512 bake gives)
#   V1  the line between the nodes' heights WITHOUT the surface detail, plus the exact 3-octave detail at every column (land and ice-sheet gate taken from the nearest node's biome): the real term, about 3 octaves of lattice noise a sample
#   V1x as V1 but the gate taken from the column's own biome (what a perfect gate would give)
#   V2  as V1 with 2 octaves (cheaper, approximate)
#   V3  a cubic (Catmull-Rom) through four nodes of the full heights: no noise, better curvature
#   V4  V0 plus a hash-of-the-column noise of the amplitude that matches V0's residual: what a 1 to 2 op "texture" term can do (it adds variance, it cannot subtract it)
# For each: RMS over all columns, over mountain columns (raw over 800), over shore columns (within 40 of SHORE_WET); and the shoreline (crossings of SHORE_WET in the raw column): displacement of the nearest variant crossing within 40
# columns (units), crossings lost, and spurious extra crossings.
func _init() -> void:
	WorldClimate.useGen(2)
	var rings: Array = [[7, 1], [3, 1], [11, 1], [25, 1], [9, 3], [7, 3], [1, 3], [5, 3]]
	var names: Array = ["V0 bilinear", "V1 +detail 3 oct", "V1x +detail exact gate", "V2 +detail 2 oct", "V3 cubic", "V4 +hash noise"]
	var sq: Array = []
	var cnt: Array = []
	var sqm: Array = []
	var cntm: Array = []
	var sqs: Array = []
	var cnts: Array = []
	var disp: Array = []
	var lost: Array = []
	var extra: Array = []
	var mx: Array = []
	for v in range(names.size()):
		sq.append(0.0)
		cnt.append(0)
		sqm.append(0.0)
		cntm.append(0)
		sqs.append(0.0)
		cnts.append(0)
		disp.append([])
		lost.append(0)
		extra.append(0)
		mx.append(0.0)
	var c: Dictionary = {}
	for rg in rings:
		var S := SimCore.createSim()
		SimCore.newMatch(S, 1, {}, {"planet": rg[0], "touch": rg[1], "arena": {"cols": 2400}, "planetGen": 2, "intro": false})
		var NC: int = SimConst.NC
		var P := WorldPlanet.make(rg[0])
		c = P.c
		var pr: Dictionary = S.planetRing
		var ring: Dictionary = WorldRingPick.ringFrom(pr.c, int(pr.cols), pr.t)
		var raw := PackedFloat64Array()
		var h0 := PackedFloat64Array()
		var det3 := PackedFloat64Array()
		var det2 := PackedFloat64Array()
		var bio := PackedInt32Array()
		raw.resize(NC)
		h0.resize(NC)
		det3.resize(NC)
		det2.resize(NC)
		bio.resize(NC)
		for i in range(NC):
			var p: Array = WorldPlanet.columnPoint(P, ring, i)
			WorldPlanet.surfAt(P, p[0], p[1], p[2])
			h0[i] = P.outH
			bio[i] = P.outB
			var seedD: int = P.seed + int(c.sDetail)
			var d3: float = (WorldPlanet._fbm(P.R, p[0], p[1], p[2], c.detailScale, int(c.detailOct), seedD) - 0.5) * c.detailAmp
			var d2: float = (WorldPlanet._fbm(P.R, p[0], p[1], p[2], c.detailScale, 2, seedD) - 0.5) * c.detailAmp
			det3[i] = d3
			det2[i] = d2
			var r: float = h0[i]
			if bio[i] >= WorldPlanet.BIO_PLAINS and bio[i] <= WorldPlanet.BIO_MOUNTAINS:
				r += d3
			elif bio[i] == WorldPlanet.BIO_ICE_SHEET and h0[i] > 100.0:
				r += d3 * c.detailSheet
			raw[i] = r
		var vs: Array = []
		for v in range(names.size()):
			var a := PackedFloat64Array()
			a.resize(NC)
			vs.append(a)
		var k: int = 16
		var resid: float = 0.0
		for i in range(NC):
			var a0: int = (i / k) * k
			var t: float = float(i - a0) / float(k)
			var n0: int = a0 % NC
			var n1: int = (a0 + k) % NC
			var nm: int = posmod(a0 - k, NC)
			var n2: int = (a0 + 2 * k) % NC
			var full0: float = raw[n0]
			var full1: float = raw[n1]
			vs[0][i] = full0 * (1.0 - t) + full1 * t
			var base: float = h0[n0] * (1.0 - t) + h0[n1] * t
			var nb: int = bio[n0] if t < 0.5 else bio[n1]
			var gate: bool = (nb >= WorldPlanet.BIO_PLAINS and nb <= WorldPlanet.BIO_MOUNTAINS) or (nb == WorldPlanet.BIO_ICE_SHEET and base > 100.0)
			var sheet: float = c.detailSheet if nb == WorldPlanet.BIO_ICE_SHEET else 1.0
			vs[1][i] = base + (det3[i] * sheet if gate else 0.0)
			var gx: bool = (bio[i] >= WorldPlanet.BIO_PLAINS and bio[i] <= WorldPlanet.BIO_MOUNTAINS) or (bio[i] == WorldPlanet.BIO_ICE_SHEET and base > 100.0)
			vs[2][i] = base + (det3[i] * (c.detailSheet if bio[i] == WorldPlanet.BIO_ICE_SHEET else 1.0) if gx else 0.0)
			vs[3][i] = base + (det2[i] * sheet if gate else 0.0)
			var y0: float = raw[nm]
			var y3: float = raw[n2]
			vs[4][i] = 0.5 * ((2.0 * full0) + (-y0 + full1) * t + (2.0 * y0 - 5.0 * full0 + 4.0 * full1 - y3) * t * t + (-y0 + 3.0 * full0 - 3.0 * full1 + y3) * t * t * t)
			resid += (vs[0][i] - raw[i]) * (vs[0][i] - raw[i])
		var amp: float = sqrt(3.0 * resid / float(NC))
		for i in range(NC):
			var hv: int = (i * 2654435761 + int(rg[0]) * 40503) & 0xffff
			vs[5][i] = vs[0][i] + (float(hv) / 32768.0 - 1.0) * amp
		for v in range(names.size()):
			for i in range(NC):
				var e: float = vs[v][i] - raw[i]
				sq[v] += e * e
				cnt[v] += 1
				mx[v] = maxf(mx[v], absf(e))
				if raw[i] > 800.0:
					sqm[v] += e * e
					cntm[v] += 1
				if absf(raw[i] - WorldWater.SHORE_WET) < 40.0:
					sqs[v] += e * e
					cnts[v] += 1
			var xr: int = 0
			for i in range(NC - 1):
				var cr: bool = (raw[i] - WorldWater.SHORE_WET) * (raw[i + 1] - WorldWater.SHORE_WET) < 0.0
				var cv: bool = (vs[v][i] - WorldWater.SHORE_WET) * (vs[v][i + 1] - WorldWater.SHORE_WET) < 0.0
				if cr:
					var best: float = 1.0e9
					for d in range(-40, 41):
						var j: int = posmod(i + d, NC)
						var j2: int = posmod(i + d + 1, NC)
						if (vs[v][j] - WorldWater.SHORE_WET) * (vs[v][j2] - WorldWater.SHORE_WET) < 0.0:
							best = minf(best, absf(float(d)) * SimConst.COL)
					if best > 1.0e8:
						lost[v] += 1
					else:
						disp[v].append(best)
				elif cv:
					var near: bool = false
					for d in range(-40, 41):
						var j: int = posmod(i + d, NC)
						if (raw[j] - WorldWater.SHORE_WET) * (raw[posmod(j + 1, NC)] - WorldWater.SHORE_WET) < 0.0:
							near = true
							break
					if not near:
						extra[v] += 1
	for v in range(names.size()):
		var ds: Array = disp[v]
		var sm: float = 0.0
		var dm: float = 0.0
		for x in ds:
			sm += x * x
			dm = maxf(dm, x)
		print("%-24s all RMS %.1f (max %.0f) | mountain RMS %.1f | shore columns RMS %.1f | shoreline: %d crossings, displacement RMS %.0f max %.0f, %d lost, %d spurious extra" % [names[v], sqrt(sq[v] / maxf(1.0, cnt[v])), mx[v], sqrt(sqm[v] / maxf(1.0, cntm[v])), sqrt(sqs[v] / maxf(1.0, cnts[v])), ds.size() + lost[v], sqrt(sm / maxf(1.0, ds.size())), dm, lost[v], extra[v]])
	quit()
