## The campaign's arc and episode (campaign-core.md): the data ships off; the seeds pinned; a filler arc's families
## and villain; a whole episode, five columns, played twice and the same to the bit; each place flown to one of its
## column's, the rise's untaken place back at the escalation, the climax at the villain's home; the run state through
## JSON; a place not of its column refused; a main-story episode's card and the season record (mc02), and its side
## story's n.5 on a copy of the planet; threads as ids, filler never resolving a main-story one; and no fighter,
## director or input file naming the campaign (Legal mc01, mc06).
func _campaign() -> String:
	SimCampaign.reload()
	if not SimCampaign.errors().is_empty():
		return "the campaign's data: %s" % str(SimCampaign.errors())
	if SimCampaign.enabled():
		return "the campaign is on in the data (it ships off)"
	var arcId: String = "c0ffee00c0ffee00"
	var pins: Array = [SimCampaign.episodeSeed(arcId, 0), SimCampaign.episodeRun(arcId, 0), SimCampaign.arcPlanet(arcId), SimCampaign.mix(7, "column", 2, 1)]
	if str(pins) != str([2515801561, "95f415d9fb019b1a", 2172332729, 3808475095]):
		return "the campaign's seeds: %s (want the pinned four: a port gives the same)" % str(pins)
	var arc: Dictionary = SimCampaign.newArc(arcId, "filler_arc")
	if arc.families.size() != 2 or arc.families[0] == arc.families[1] or not SimCampaign.families.has(arc.families[0]) or not SimCampaign.families.has(arc.families[1]) or arc.villain != "RIVAL" or SimCampaign.checkArc(arc) != "" or SimCampaign.cardNumber(arc) != "":
		return "a filler arc: families %s, villain %s, check '%s', card '%s' (want two different filler families, the RIVAL, sound, no number)" % [str(arc.families), arc.villain, SimCampaign.checkArc(arc), SimCampaign.cardNumber(arc)]
	# a whole episode, twice
	var e1: Dictionary = SimCampaignRun.play(arcId, "filler_arc", 240)
	var e2: Dictionary = SimCampaignRun.play(arcId, "filler_arc", 240)
	if JSON.stringify(e1.stages) != JSON.stringify(e2.stages) or e1.run != e2.run or JSON.stringify(e1.arc) != JSON.stringify(e2.arc):
		return "an episode played twice differs"
	var a1: Dictionary = e1.arc
	var n: int = SimCampaign.columns
	if not a1.done or a1.taken.size() != n or a1.columns.size() != n:
		return "an episode: done %s, %d taken, %d columns (want done, %d and %d)" % [str(a1.done), a1.taken.size(), a1.columns.size(), n, n]
	var P: WorldPlanet.Planet = WorldPlanet.make(SimCampaign.arcPlanet(arcId))
	for c in range(n):
		var col: Array = a1.columns[c]
		if not SimCampaign._has(col, int(a1.taken[c])) or (c < n - 1 and (col.size() < SimCampaign.placesMin or col.size() > SimCampaign.placesMax)):
			return "column %d: places %s, flown to %d (want %d to %d places, the one flown to among them)" % [c + 1, str(col), int(a1.taken[c]), SimCampaign.placesMin, SimCampaign.placesMax]
		for k in range(c):
			if int(a1.taken[k]) == int(a1.taken[c]):
				return "the episode fought on place %d twice" % int(a1.taken[c])
	var home: int = int(SimRun.homes(P).get("RIVAL", -1))
	if a1.columns[n - 1] != [home]:
		return "the climax: %s (want the villain's home, %d)" % [str(a1.columns[n - 1]), home]
	var back: int = -1
	for i in a1.columns[SimCampaign.crossFrom - 1]:
		if int(i) != int(a1.taken[SimCampaign.crossFrom - 1]):
			back = int(i)
			break
	if back < 0 or int(a1.columns[SimCampaign.crossTo - 1][0]) != back:
		return "the escalation's places %s don't open with the rise's untaken place %d" % [str(a1.columns[SimCampaign.crossTo - 1]), back]
	var sj = JSON.parse_string(JSON.stringify(e1.state))
	if SimRun.stateHash(sj) != e1.run or SimRun.checkState(sj) != "":
		return "an episode's run state through JSON: hash %s, check '%s'" % [SimRun.stateHash(sj), SimRun.checkState(sj)]
	# a place not of the column is refused, and the state is left as it was
	var quiet: bool = FighterData.quiet
	FighterData.quiet = true
	var s0: Dictionary = SimCampaign.newEpisode(arc, 0, P, {"people": 100000})
	var bad: int = 0
	while SimCampaign._has(s0.arc.columns[0], bad):
		bad += 1
	var nErr: int = SimCampaign.errors().size()
	var sr: Dictionary = SimCampaign.advance(s0, {"result": "won", "place": bad, "places": {}}, P)
	var refused: bool = SimRun.stateHash(sr) == SimRun.stateHash(s0) and SimCampaign.errors().size() == nErr + 1
	# threads: ids only; filler never resolves a main-story thread
	var st: Dictionary = SimCampaign.openThread(SimCampaign.openThread(e1.state, "grudge:RIVAL"), "main:the-pier")
	var st2: Dictionary = SimCampaign.openThread(st, "Lost 400 people")
	var threadsOk: bool = st.arc.threads == ["grudge:RIVAL", "main:the-pier"] and st2.arc.threads == st.arc.threads
	threadsOk = threadsOk and SimCampaign.closeThread(st, "main:the-pier").arc.threads == st.arc.threads and SimCampaign.closeThread(st, "grudge:RIVAL").arc.threads == ["main:the-pier"]
	FighterData.quiet = quiet
	if not refused or not threadsOk:
		return "the refusals: a place not of its column %s; threads %s (want the state left as it was, ids only, a main-story thread kept open in filler)" % [str(refused), str(threadsOk)]
	# a main-story episode: its card is its number, the season moves up when it is won, and a side story reads n.5
	var mainId: String = ""
	for id in ["5eed0000decade42", "0123456789abcdef", "00c0ffee0badf00d", "feedface00000001", "abad1dea00000002"]:
		if SimRun.H(id, "side", 12, 0) % 1000 < SimCampaign.sideShare:
			mainId = id
			break
	var m: Dictionary = SimCampaignRun.play(mainId, "main_episode", 120, {"number": 12})
	var season: Dictionary = SimCampaign.seasonAfter(SimCampaign.newSeason(), m.state)
	var extra: Dictionary = season.duplicate()
	extra["player"] = "x"
	if SimCampaign.cardNumber(m.arc) != "12" or int(season.lastMain) != 12 or SimCampaign.checkSeason(season) != "" or SimCampaign.checkSeason(extra) == "" or not m.arc.side.offered:
		return "a main-story episode: card '%s', season %s, a season with a name refused %s, side %s (want 12, lastMain 12, refused, offered)" % [SimCampaign.cardNumber(m.arc), str(season), str(SimCampaign.checkSeason(extra) != ""), str(m.arc.side)]
	var PM: WorldPlanet.Planet = WorldPlanet.make(SimCampaign.arcPlanet(mainId))
	var side: Dictionary = SimCampaign.openSide(m.state, PM)
	if side.is_empty() or SimCampaign.cardNumber(side.arc) != "12.5" or side.arc.kind != "filler_episode" or int(side.planet.seed) != int(m.state.planet.seed) or JSON.stringify(side.places) != JSON.stringify(m.state.places) or side.arc.families != [m.arc.side.family]:
		return "the side story: card '%s', kind %s, on the main arc's planet as it stands %s (want 12.5, a filler episode, yes)" % [SimCampaign.cardNumber(side.arc) if not side.is_empty() else "", side.get("arc", {}).get("kind", ""), str(not side.is_empty() and JSON.stringify(side.places) == JSON.stringify(m.state.places))]
	# no fighter, director or input file names the campaign (Legal mc01 and mc06: nothing of it reaches a fighter)
	var files: Array = []
	for dir in ["res://sim/core/", "res://sim/director/", "res://sim/input/"]:
		for f in DirAccess.get_files_at(dir):
			if f.ends_with(".gd") and not ["campaign.gd", "run.gd"].has(f):
				files.append(dir + f)
	for path in files:
		var text: String = FileAccess.get_file_as_string(path)
		if text.contains("SimCampaign") or text.contains("state.arc") or text.contains("arc_villain"):
			return "%s names the campaign: no fighter reads it (mc01, mc06)" % path
	return ""
