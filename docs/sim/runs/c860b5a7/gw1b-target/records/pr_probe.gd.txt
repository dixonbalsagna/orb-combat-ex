extends SceneTree
# pr_probe.gd: parity's shared-functions row, the second send, on the planet (the 44-line GW1b tree): predict()'s track
# against the flight, tick by tick around the touchdown; then the last stand row with RT pressed on the tick.

func _init() -> void:
	WorldDefault.data()
	var dir: Dictionary = {"ux": 0.6, "uy": 0.8}
	var M := SimCore.createSim()
	SimCore.newMatch(M, 5, {"p1": false, "p2": false}, {"flight": true})
	var A = M.fighters[0]
	var D = M.fighters[1]
	D.y = WorldTerrain.groundY(M, D.x) + 400.0
	DirLaunch.doLaunch(M, A, D, dir, 2400.0, false)
	print("start x %.2f y %.2f vx %.3f vy %.3f aimB %d floorOn %s" % [D.x, D.y, D.vx, D.vy, D.aimB, str(WorldFill.floorOn())])
	var tr = SimFlight.track(D.x, D.y, D.vx, D.vy, D.aimB)
	var trk: Dictionary = {}
	while tr.t < 300:
		SimFlight.tick(M, tr)
		trk[int(tr.t)] = [tr.x, tr.y]
	for t in range(1, 300):
		SimCore.step(M, [null, null])
		M.out.fx.clear()
		M.out.feed.clear()
		if t >= 250:
			var p: Array = trk.get(t, [0.0, 0.0])
			print("t %d flight x %.3f y %.3f vy %.3f state %s wet %s contacts %d | ground %.3f surface %.3f floorAt %.3f | track x %.3f y %.3f floor %.3f" % [t, D.x, D.y, D.vy, D.state, str(D.wet), D.jContacts, WorldTerrain.groundY(M, D.x), WorldWater.surfaceAt(M, D.x), WorldWater.floorAt(M, D.x, D.z), p[0], p[1], SimFlight._floor(M, p[0])])
		if D.state != "launched" or D.jContacts > 0 or D.slide > 0.0 or D.wet or D.flightHits > 0:
			break
	SimCore.dispose(M)
	# the last stand, RT pressed on the tick of the free signature's B
	var S := SimCore.createSim()
	SimCore.newMatch(S, 5, {"p1": false, "p2": false})
	var a = S.fighters[0]
	S.dirS.cool = 0.0
	a.ki = 0.0
	a.sigReadyT = S.T + 100.0
	a.input.stanceMask = 4
	a.input.powerPress = true
	DirExchange.requestAttack(S, a, "sig")
	print("last stand, no ki, RT pressed: ex %s" % str(S.dirS.ex != null))
	a.lastStandUsed = true
	a.lastStandLeft = 600
	S.out.fx.clear()
	DirExchange.requestAttack(S, a, "sig")
	var used: int = 0
	for e in S.out.fx:
		if e.type == "last_stand_end" and e.kind == "used":
			used += 1
	print("last stand, free, RT pressed: ex %s kind %s ki %s left %d used %d cooldown %s" % [str(S.dirS.ex != null), str(S.dirS.ex.kind if S.dirS.ex != null else "-"), str(a.ki), a.lastStandLeft, used, str(a.sigReadyT > S.T)])
	SimCore.dispose(S)
	quit(0)
