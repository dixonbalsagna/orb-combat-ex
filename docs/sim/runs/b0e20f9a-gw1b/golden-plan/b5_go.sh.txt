#!/bin/sh
# b5_go.sh <scratchpad>: the GW1b golden plan's recipe change and arena pins (kick/gw1b_golden_edit.py) on b4 (steps 1-4, arena)
# and b4f (+ the flip). ONE Godot job at a time, below normal (a long run).
set -u
powershell -NoProfile -Command "(Get-Process -Id $(cat /proc/$$/winpid)).PriorityClass = 'BelowNormal'"
G="/c/Users/<user>/AppData/Local/Programs/Godot/4.7.2/Godot_v4.7.2-stable_win64_console.exe"
GW="C:/Users/<user>/AppData/Local/Programs/Godot/4.7.2/Godot_v4.7.2-stable_win64_console.exe"
NS="${1:?scratchpad}"; O="$NS/b5o"
C="$NS/b5app/Godot/app_userdata/Orb Combat EX/field-cache"
rm -rf "${NS:?}/b5o"; mkdir -p "$O" "$NS/b5app"; export APPDATA="$(cygpath -w "$NS/b5app")"
run() { name="$1"; t="$2"; shift 2; s=$(date +%s); timeout "$t" "$G" --headless --path . --script "$@" > "$O/$name.txt" 2>&1; code=$?; echo "$name: exit $code in $(( $(date +%s) - s )) s; script errors $(grep -c 'SCRIPT ERROR\|Parse Error' "$O/$name.txt"); $(grep -v 'WARNING\|at: \|^$\|ERROR: \|wait_to_finish' "$O/$name.txt" | tail -1 | cut -c1-150)"; grep -n "^FAIL\|^  FAIL" "$O/$name.txt" | head -12 | cut -c1-260; }
echo "== A. steps 1-4 (arena) + the recipe change and pins"
cd "$NS/b4" || exit 1
case "$(pwd -P)" in */scratchpad/b4) ;; *) echo refusing; exit 1 ;; esac
git -c core.autocrlf=false diff -- sim/core/tools > "$O/gw1b-golden.patch"
echo "gw1b-golden sha1 $(sha1sum "$O/gw1b-golden.patch" | cut -c1-40) sha256 $(sha256sum "$O/gw1b-golden.patch" | cut -c1-12) $(grep -c '^diff --git' "$O/gw1b-golden.patch") files"
timeout 900 "$G" --headless --path . --import > "$O/import_a.txt" 2>&1; echo "import $?; parse errors $(grep -c 'Parse Error' "$O/import_a.txt")"
cp sim/core/test/golden.json "$O/golden_main.json"
run golden_a 1200 res://sim/core/tools/golden.gd
cp sim/core/test/golden.json "$O/golden_a.json"
s=$(date +%s); node tools/gate.mjs -- "$GW" --headless --path . --script res://sim/core/tools/parity.gd > "$O/parity_a.txt" 2>&1; echo "parity, arena + recipes (its regenerated golden): exit $? in $(( $(date +%s) - s )) s; $(grep 'golden check\|SCRIPT ERROR lines' "$O/parity_a.txt" | tail -2 | tr '\n' ' ' | cut -c1-160)"
grep -n "^FAIL\|planet match\|slices" "$O/parity_a.txt" | head -8 | cut -c1-220
cp "$O/golden_main.json" sim/core/test/golden.json
for c in frame_check cross_check cross_end_check fights_check; do run "a_$c" 2400 "res://sim/core/tools/$c.gd"; done
echo "== B. + the flip (planet)"
cd "$NS/b4f" || exit 1
case "$(pwd -P)" in */scratchpad/b4f) ;; *) echo refusing; exit 1 ;; esac
timeout 900 "$G" --headless --path . --import > "$O/import_b.txt" 2>&1; echo "import $?; parse errors $(grep -c 'Parse Error' "$O/import_b.txt")"
rm -rf "${C:?}"
run golden_b_cold 2400 res://sim/core/tools/golden.gd
cp sim/core/test/golden.json "$O/golden_b_cold.json"
run golden_b_warm 2400 res://sim/core/tools/golden.gd
cmp -s sim/core/test/golden.json "$O/golden_b_cold.json" && echo "  the golden made twice, a cold cache then a warm one: ONE file" || echo "  the golden made twice: the files DIFFER"
s=$(date +%s); node tools/gate.mjs -- "$GW" --headless --path . --script res://sim/core/tools/parity.gd > "$O/parity_b.txt" 2>&1; echo "parity, the flip + recipes + pins (its regenerated golden): exit $? in $(( $(date +%s) - s )) s"
grep -n "^FAIL" "$O/parity_b.txt" | head -12 | cut -c1-260
grep -n "matches\|planet match" "$O/parity_b.txt" | grep "ok\|identical" | head -4 | cut -c1-200
cp "$O/golden_main.json" sim/core/test/golden.json
for c in frame_check cross_check cross_end_check fights_check; do run "b_$c" 2400 "res://sim/core/tools/$c.gd"; done
echo b5 done
