#  - the recipes: two of the nine computer matches (default seed 1, and seed 11's KO) and both scripted replays take
#    "world": "arena", so the hand-made arena keeps a reference; the other seven play the round planet once the flip is in;
#    PLANET_MATCH names generation 3 (planet 7, touch 3, gen 3: the world Orb played) and records its first crater and first
#    slide, which parity requires (the GW1a review's S-07; path-round-2's "a crater and a slide");
#  - parity re-runs each match and replay with its recorded set-up;
#  - my rows that are about the hand-made arena run in arena mode (GW1b review B-02): nine of parity's (not the composed
#    intros: a real fault on the planet), and cross_check, cross_end_check and fights_check whole (their stages are written on
#    the arena); frame_check's two "off a field world" matches take "world": "arena".
import sys, os

T = sys.argv[1]
assert os.path.isdir(os.path.join(T, ".git")) and "scratchpad" in os.path.abspath(T).replace("\\", "/"), "refusing"


def edit(rel, pairs):
    p = os.path.join(T, rel)
    s = open(p, encoding="utf-8", newline="").read()
    for a, b in pairs:
        n = s.count(a)
        assert n == 1, "%s: anchor found %d times: %r" % (rel, n, a[:90])
        s = s.replace(a, b, 1)
    open(p, "w", encoding="utf-8", newline="").write(s)
    print("edited", rel)
