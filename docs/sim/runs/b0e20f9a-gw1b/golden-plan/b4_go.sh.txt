#!/bin/sh
# b4_go.sh <scratchpad>: GW1b's golden plan, on main b0e20f9a.
#   A. b4 = main + steps 1-4 (arena mode): the golden byte for byte main's; parity (replay-world's row); world_default_check; the gen-3 recipe probe.
#   B. b4f = b4 + the flip (data-switches-gw1b.diff, mode planet): the golden (moves), parity against it, and my CI checks: the rows that go red on the planet.
# ONE Godot job at a time, below normal (a long run).
set -u
powershell -NoProfile -Command "(Get-Process -Id $(cat /proc/$$/winpid)).PriorityClass = 'BelowNormal'"
G="/c/Users/<user>/AppData/Local/Programs/Godot/4.7.2/Godot_v4.7.2-stable_win64_console.exe"
GW="C:/Users/<user>/AppData/Local/Programs/Godot/4.7.2/Godot_v4.7.2-stable_win64_console.exe"
NS="${1:?scratchpad}"; O="$NS/b4o"
R="/c/Users/<user>/Documents/Claude Proj/Breakers-Like"
rm -rf "${NS:?}/b4o"; mkdir -p "$O" "$NS/b4app"; export APPDATA="$(cygpath -w "$NS/b4app")"
run() { name="$1"; t="$2"; shift 2; s=$(date +%s); timeout "$t" "$G" --headless --path . --script "$@" > "$O/$name.txt" 2>&1; code=$?; echo "$name: exit $code in $(( $(date +%s) - s )) s; script errors $(grep -c 'SCRIPT ERROR\|Parse Error' "$O/$name.txt"); $(grep -v 'WARNING\|at: \|^$\|ERROR: \|wait_to_finish' "$O/$name.txt" | tail -1 | cut -c1-150)"; grep -n "^FAIL\|^  FAIL" "$O/$name.txt" | head -12 | cut -c1-260; }
echo "== A. steps 1-4, arena mode"
cd "$NS/b4" || exit 1
case "$(pwd -P)" in */scratchpad/b4) ;; *) echo refusing; exit 1 ;; esac
timeout 900 "$G" --headless --path . --import > "$O/import.txt" 2>&1; echo "import $?; parse errors $(grep -c 'Parse Error' "$O/import.txt")"
cp sim/core/test/golden.json "$O/golden_main.json"
run golden_a 900 res://sim/core/tools/golden.gd
cmp -s sim/core/test/golden.json "$O/golden_main.json" && echo "  steps 1-4: golden.json byte-identical to main's (expected: arena mode)" || echo "  steps 1-4: golden.json MOVED (unexpected)"
cp "$O/golden_main.json" sim/core/test/golden.json
s=$(date +%s); node tools/gate.mjs -- "$GW" --headless --path . --script res://sim/core/tools/parity.gd > "$O/parity_a.txt" 2>&1; echo "parity, steps 1-4: exit $? in $(( $(date +%s) - s )) s; $(grep 'golden check\|SCRIPT ERROR lines' "$O/parity_a.txt" | tail -2 | tr '\n' ' ' | cut -c1-200)"
grep -n "^FAIL\|default world names its planet" "$O/parity_a.txt" | head -6 | cut -c1-200
run world_default 1200 res://sim/world/tools/world_default_check.gd
run gen3_probe 3600 res://sim/core/tools/gen3_probe.gd
grep "^seed" "$O/gen3_probe.txt"
echo "== B. steps 1-4 + the flip (mode planet)"
rm -rf "${NS:?}/b4f"; cp -r "$NS/b4" "$NS/b4f" || exit 1
cd "$NS/b4f" || exit 1
case "$(pwd -P)" in */scratchpad/b4f) ;; *) echo refusing; exit 1 ;; esac
git -c core.autocrlf=false apply "$R/docs/world/pending/a-world-01/data-switches-gw1b.diff" || { echo "the flip did not apply"; exit 1; }
grep -n '"mode"' data/biomes/world.json | head -2
run golden_b 2400 res://sim/core/tools/golden.gd
cmp -s sim/core/test/golden.json "$O/golden_main.json" && echo "  the flip: golden.json byte-identical (unexpected)" || echo "  the flip: golden.json MOVED (expected)"
cp sim/core/test/golden.json "$O/golden_flip.json"
s=$(date +%s); node tools/gate.mjs -- "$GW" --headless --path . --script res://sim/core/tools/parity.gd > "$O/parity_b.txt" 2>&1; echo "parity, the flip (its regenerated golden): exit $? in $(( $(date +%s) - s )) s"
grep -n "^FAIL" "$O/parity_b.txt" | head -30 | cut -c1-260
cp "$O/golden_main.json" sim/core/test/golden.json
s=$(date +%s); node tools/gate.mjs --verdict -- "$GW" --headless --path . --script res://sim/core/tools/power_check.gd > "$O/b_power.txt" 2>&1; echo "power_check, the flip: exit $? in $(( $(date +%s) - s )) s; $(grep 'POWER' "$O/b_power.txt" | tail -1)"; grep -n "^FAIL" "$O/b_power.txt" | head -8 | cut -c1-200
for c in cross_check cross_end_check field_static_check fork_check frame_check kick_check load_split memo_check places_check rec_check replay_xproc_check ceil_check fights_check coop_check loop_check; do run "b_$c" 2400 "res://sim/core/tools/$c.gd"; done
run b_two_states 2400 res://sim/core/tools/two_states_check.gd -- --setups=default,power,dash,slip,fights
echo b4 done
