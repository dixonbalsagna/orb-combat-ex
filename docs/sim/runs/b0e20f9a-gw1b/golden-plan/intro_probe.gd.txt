extends SceneTree
## Scratch probe (Simulation, 2026-11-21): on planet 7 (the flip), what differs at the clock between a composed intro played out
## and the setup's skip. Never to land.

func _snap(S: SimState) -> Dictionary:
	var d := {}
	for k in range(2):
		var f = S.fighters[k]
		d["f%d" % k] = [f.x, f.y, f.z, f.vx, f.vy, f.state, f.hp, f.ki, f.power, f.face]
	var cr: Array = []
	for c in S.craters:
		cr.append([c.x, c.r if c.get("r") != null else 0.0, c.d if c.get("d") != null else 0.0])
	d["craters"] = cr
	var h := SimHash.Hasher.new()
	for v in S.deform:
		h.num(float(v))
	d["deform"] = h.hex()
	var hb := SimHash.Hasher.new()
	for b in S.buildings:
		hb.num(float(b.hp)); hb.num(float(b.popAlive))
	d["buildings"] = hb.hex()
	d["seed"] = S.seed if S.get("seed") != null else -1
	d["rng"] = S.rng.a
	var recs: Array = []
	for i in range(int(S.storeN)):
		var o: int = i * WorldStore.STRIDE
		recs.append([S.store[o], snappedf(S.store[o + 1], 0.000001), snappedf(S.store[o + 4], 0.001), S.store[o + 8]])
	d["storeRecs"] = recs
	d["T"] = S.T
	var keep: Array = [S.tick, S.intro.t, S.intro.scenario, S.intro.first, S.intro.gap, S.intro.picks, S.intro.clock]
	S.tick = 0
	S.intro.t = 0
	S.intro.scenario = -1
	S.intro.first = 0
	S.intro.gap = 0
	S.intro.picks = 0
	S.intro.clock = 0
	d["clockHash"] = SimHash.stateHash(S).gameplay
	d["vec"] = SimHash.collect(S, "gameplay")
	S.tick = keep[0]
	S.intro.t = keep[1]
	S.intro.scenario = keep[2]
	S.intro.first = keep[3]
	S.intro.gap = keep[4]
	S.intro.picks = keep[5]
	S.intro.clock = keep[6]
	for key in ["epoch", "swing", "frame", "hs", "cata", "mood"]:
		var o = S.get(key)
		if o == null:
			d[key] = "null"
		elif o is Object:
			var parts: Array = []
			for pr in o.get_property_list():
				if pr.usage & PROPERTY_USAGE_SCRIPT_VARIABLE:
					var v = o.get(pr.name)
					if not (v is Object):
						parts.append("%s=%s" % [pr.name, str(v).substr(0, 60)])
			d[key] = ", ".join(parts)
		else:
			d[key] = str(o).substr(0, 200)
	return d


func _init() -> void:
	var quiet := SimIntent.new()
	var Q := SimCore.createSim()
	SimCore.newMatch(Q, 5, {"p1": false, "p2": false}, {"intro": "skip"})
	var a: Dictionary = _snap(Q)
	print("skip at the clock: planet %s, %s" % [str(Q.get("planetSeed")), str(a)])
	SimCore.dispose(Q)
	var T: Array = SimIntro.templates
	var ti: int = 0
	for k in range(T.size()):
		if str(T[k].id) == "double_drop":
			ti = k
	var tp: Dictionary = T[ti]
	var rec := {"play": true, "scenario": tp.id, "order": 1, "gap": 30, "facts": {}}
	var S := SimCore.createSim()
	SimCore.newMatch(S, 5, {"p1": false, "p2": false}, {"intro": rec})
	for t in range(tp.clock):
		SimCore.step(S, [quiet, quiet])
	var b: Dictionary = _snap(S)
	print("played at the clock: %s" % str(b))
	var va: Array = a.vec
	var vb: Array = b.vec
	print("vector sizes: skip %d, played %d" % [va.size(), vb.size()])
	var nd: int = 0
	for i in range(mini(va.size(), vb.size())):
		if str(va[i]) != str(vb[i]):
			nd += 1
			if nd <= 12:
				var ctx: Array = []
				for j in range(maxi(0, i - 3), i):
					ctx.append(str(va[j]).substr(0, 30))
				print("VEC %d: skip %s | played %s   (before: %s)" % [i, str(va[i]).substr(0, 60), str(vb[i]).substr(0, 60), str(ctx)])
	print("vector entries differing: %d" % nd)
	a.erase("vec")
	b.erase("vec")
	for key in a:
		if str(a[key]) != str(b.get(key)):
			print("DIFFERS %s: skip %s | played %s" % [key, str(a[key]), str(b.get(key))])
	SimCore.dispose(S)
	quit(0)
