#!/usr/bin/env python3
# gw1b_energy_edit.py <tree>: Game Design's addendum for GW1b (docs/design/energy-stamina-body.md section 6, flight-3500.md),
# on top of close-rows, energy-on, the gates park and meters-zero.
#  - the grant into data/fight/power.json: flight.flatOut 3,500 (A-CONTROLS-04), dash.closeSec {1000: 0.45, 3000: 1.0,
#    6000: 1.9}, launchTravel 1.5 (Orb's 9 November match file: ?travel=1.5);
#  - power_check's rows keyed to the old 5,000 and x2 read the data instead; the close times' targets come from closeSec,
#    and 0.9 s at the top from 6,000 (flight-3500.md);
#  - Narrative's grant: every feed to the meters is 0 (destroyed, destroyed_power, emptied, crater, failed_to_protect, region_broken,
#    brink: Game Design's ruling of 2026-11-14), each with a note of what it was.
import sys, os, json

T = sys.argv[1]
assert os.path.isdir(os.path.join(T, ".git")) and "scratchpad" in os.path.abspath(T).replace("\\", "/"), "refusing"


def edit(rel, pairs):
    p = os.path.join(T, rel)
    s = open(p, encoding="utf-8", newline="").read()
    for a, b in pairs:
        n = s.count(a)
        assert n == 1, "%s: anchor found %d times: %r" % (rel, n, a[:90])
        s = s.replace(a, b, 1)
    open(p, "w", encoding="utf-8", newline="").write(s)
    print("edited", rel)


edit("data/fight/power.json", [
    ('"flight": {"flatOut": 5000.0,', '"flight": {"flatOut": 3500.0,'),
    ('"closeSec": {"1000": 0.4, "3000": 0.8, "6000": 1.4}', '"closeSec": {"1000": 0.45, "3000": 1.0, "6000": 1.9}'),
    ('  "launchTravel": 2.0,\n', '  "launchTravel": 1.5,\n'),
    ('"_flight": "flat out, plain stance: flatOut x growth^level.',
     '"_flight": "flat out, plain stance: flatOut x growth^level. flatOut 3,500 since GW1b (Orb\'s decision 5, Game Design\'s flight-3500.md; it was 5,000).'),
    ("Game Design's lever is 1.5, then 1.25 (slam-and-pace.md section 2)",
     "Game Design's lever is 1.5, then 1.25 (slam-and-pace.md section 2). launchTravel 1.5 since GW1b: Orb's 9 November match ran ?travel=1.5 (flight-3500.md)"),
])
j = json.load(open(os.path.join(T, "data/fight/power.json"), encoding="utf-8"))
assert j["flight"]["flatOut"] == 3500.0 and j["launchTravel"] == 1.5 and j["dash"]["closeSec"]["3000"] == 1.0

edit("sim/core/tools/power_check.gd", [
    ('\t\tok(near(far, row[5] * f.spd, 25.0), "flat out: %.0f against %.0f" % [far, row[5] * f.spd])\n',

     '\t\tok(near(far, wantFar, 25.0), "flat out: %.0f against %.0f" % [far, wantFar])\n'),
    ('\tok(near(WorldSlide.launchTravel(1.0, 0.0), 2.0, 1.0e-6) and WorldSlide.launchTravel(0.0, 1.0) == 1.0, "along the line, twice; straight up, once")\n',
     '\tok(near(WorldSlide.launchTravel(1.0, 0.0), SimPower.launchTravel, 1.0e-6) and WorldSlide.launchTravel(0.0, 1.0) == 1.0, "along the line, the data\'s x%.2f; straight up, once" % SimPower.launchTravel)\n'),
    ('\t\tvar want: float = 5000.0 * SimDetMath.pow(2.305, got[0][2]) * got[0][3]\n',
     '\t\tvar want: float = SimPower.flatOut * SimDetMath.pow(2.305, got[0][2]) * got[0][3]\n'),

    ('and topMax > 11000.0,', 'and topMax > 2.0 * SimPower.flatOut,'),

     '\t\tvar cs: Dictionary = SimPower._d.get("dash", {}).get("closeSec", {})   # QA\'s targets: power.json dash.closeSec (Game Design\'s ruling (c), moved for 3,500 by flight-3500.md)\n'
     '\t\tfor row in [[1000.0, 1.0, 0.0, float(cs.get("1000", 0.45))], [3000.0, 1.0, 0.0, float(cs.get("3000", 1.0))], [6000.0, 1.0, 0.0, float(cs.get("6000", 1.9))], [6000.0, 4.0, 100.0, 0.9]]:   # at the top from 6,000: 0.9 s (flight-3500.md)\n'),
])

Z = ("0 since Game Design's addendum to Orb's decision 6 (docs/design/energy-stamina-body.md section 6, 2026-11-14): every feed "
     "to the meters is 0, so every fighter's ladder fills alike from time, charging and damage. It was %s")


def zero(rel, rows):
    p = os.path.join(T, rel)
    s = open(p, encoding="utf-8", newline="").read()
    for old, was in rows:
        assert s.count(old) == 1, (rel, old[:60])
        a = '"amount": ' + was + ','
        assert old.count(a) == 1, (rel, a)
        s = s.replace(old, old.replace(a, '"amount": 0,\n          "_zeroed": "' + Z % was + '",'), 1)
    open(p, "w", encoding="utf-8", newline="").write(s)
    d = json.load(open(p, encoding="utf-8"))
    print("edited", rel, [(x["event"], x["amount"]) for m in d["meters"].values() for x in m.get("sources", [])])

print("ok")
