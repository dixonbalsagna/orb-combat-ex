#!/usr/bin/env python3
# gw1b_tests_fix.py <tree>: after gw1b_energy_edit.py, the two rows its first gate found written for the old numbers:
#  - parity's WIRED rows for anguish source 0's "within", menace source 2's "minBh" and source 3's "sizes": each only matters
#    while its source's amount is not 0, and since Game Design's addendum every collateral feed's amount is 0 (as the caps
#    came out in meters-zero). The amount rows stay: they prove each feed is still wired;
#  - power_check's slip row "off: a tap is still at ..." asked for twice the slip's speed, written for the 5,000 base; it now
#    asks that, off, the tap is still a third of the way from the slip's speed to his last held speed.
import sys, os

T = sys.argv[1]
assert os.path.isdir(os.path.join(T, ".git")) and "scratchpad" in os.path.abspath(T).replace("\\", "/"), "refusing"


def edit(rel, pairs):
    p = os.path.join(T, rel)
    s = open(p, encoding="utf-8", newline="").read()
    for a, b in pairs:
        n = s.count(a)
        assert n == 1, "%s: anchor found %d times: %r" % (rel, n, a[:90])
        s = s.replace(a, b, 1)
    open(p, "w", encoding="utf-8", newline="").write(s)
    print("edited", rel)

     "## (2026-11-13), are regen_penalty's and regen_bonus's caps: they only matter while their perPoint is not 0, and both are 0 in data.\n"
     "## And since Game Design's addendum (2026-11-14), anguish source 0's within, menace source 2's minBh and source 3's sizes: each\n"
     "## only matters while its source's amount is not 0, and every collateral feed's amount is 0 in data. Their amount rows stay.\n"),
    ('\t["PROTAGONIST/meters.json", ["meters", "anguish", "sources", 0, "within"], 10.0, "aWithin"],\n', ''),
