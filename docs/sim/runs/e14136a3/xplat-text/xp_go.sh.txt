#!/bin/sh
# xp_go.sh <scratchpad>: the cross-platform text fix on main (HEAD): edit, import, the golden from a cold cache, the golden
# against main's (only matches 5 and 6's light digests may move, to CI's Linux values), parity through gate.mjs.
# ONE Godot job at a time, below normal.
set -u
powershell -NoProfile -Command "(Get-Process -Id $(cat /proc/$$/winpid)).PriorityClass = 'BelowNormal'"
G="/c/Users/<user>/AppData/Local/Programs/Godot/4.7.2/Godot_v4.7.2-stable_win64_console.exe"
GW="C:/Users/<user>/AppData/Local/Programs/Godot/4.7.2/Godot_v4.7.2-stable_win64_console.exe"
NS="${1:?scratchpad}"; O="$NS/xpo"
C="$NS/xpapp/Godot/app_userdata/Orb Combat EX/field-cache"
rm -rf "${NS:?}/xpo"; mkdir -p "$O" "$NS/xpapp"; export APPDATA="$(cygpath -w "$NS/xpapp")"
[ -e "$NS/x2" ] && { echo "x2 exists"; exit 1; }
sh "$NS/mkbase3.sh" "$NS/x2" || exit 1
cd "$NS/x2" || exit 1
case "$(pwd -P)" in */scratchpad/x2) ;; *) echo refusing; exit 1 ;; esac
python -I "$NS/kick/xplat_edit.py" "$NS/x2" > "$O/edit.txt" 2>&1 || { echo "edit failed"; cat "$O/edit.txt"; exit 1; }
git -c core.autocrlf=false diff > "$O/xplat.diff"
cp sim/core/test/golden.json "$O/golden_main.json"
timeout 900 "$G" --headless --path . --import > "$O/import.txt" 2>&1; echo "import $?; parse errors $(grep -c 'Parse Error' "$O/import.txt")"
rm -rf "${C:?}"
s=$(date +%s); timeout 2400 "$G" --headless --path . --script res://sim/core/tools/golden.gd > "$O/golden.txt" 2>&1; echo "golden: exit $? in $(( $(date +%s) - s )) s; $(grep -c 'SCRIPT ERROR' "$O/golden.txt") script errors"
cp sim/core/test/golden.json "$O/golden_new.json"
python -I "$NS/kick/golden_diff.py" "$O/golden_main.json" "$O/golden_new.json"
s=$(date +%s); node tools/gate.mjs -- "$GW" --headless --path . --script res://sim/core/tools/parity.gd > "$O/parity.txt" 2>&1; echo "parity: exit $? in $(( $(date +%s) - s )) s; $(grep 'golden check\|SCRIPT ERROR lines' "$O/parity.txt" | tail -2 | tr '\n' ' ' | cut -c1-200)"
grep -n "^FAIL" "$O/parity.txt" | head -10 | cut -c1-240
grep -n "every platform\|planet worlds\|matches:" "$O/parity.txt" | head -5 | cut -c1-200
echo "rows ok $(grep -c '^ok' "$O/parity.txt"), FAIL $(grep -c '^FAIL' "$O/parity.txt")"
echo xp done
