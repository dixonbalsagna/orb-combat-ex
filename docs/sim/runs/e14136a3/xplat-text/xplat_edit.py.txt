#!/usr/bin/env python3
# xplat_edit.py <tree>: CI red on GW1b (run 45046244728): text the golden hashes read differently on Windows.
#  - mine, sim/core/mathx.gd: SimMathx.fixed(v, n), a float as n-decimal text that is the same on every platform: printf
#    for every value but an exact half, which it rounds to even (glibc's and musl's rule; the Windows build's printf rounds
#    it away from zero);
#  - Encounter's (grant), sim/director/swing.gd: the swing planner's _n() uses it;
#  - mine, golden_recipes.gd: the matches and replays as build() runs them in one place each (matchEntry, replayEntry,
#    REPLAYS), so web_parity.gd runs the same; planetSlices() names its generation (2, the golden's) instead of reading
#    the last match's (a planet match leaves 3);
#  - mine, web_parity.gd: the golden's keys through those;
#  - mine, parity.gd: a row for the halves (SimMathx.fixed and the swing line), and the planet row reads the slices with a
#    planet match's generation left in force too.
import sys, os

T = sys.argv[1]
assert os.path.isdir(os.path.join(T, ".git")) and "scratchpad" in os.path.abspath(T).replace("\\", "/"), "refusing"


def edit(rel, pairs):
    p = os.path.join(T, rel)
    s = open(p, encoding="utf-8", newline="").read()
    for a, b in pairs:
        n = s.count(a)
        assert n == 1, "%s: anchor found %d times: %r" % (rel, n, a[:90])
        s = s.replace(a, b, 1)
    open(p, "w", encoding="utf-8", newline="").write(s)
    print("edited", rel)


edit("sim/core/mathx.gd", [
    ("## A float64 from its bit pattern, 16 hex digits with the high word first.",
     "## A float as text with n decimals (0 to 6), the same on every platform. printf (\"%.nf\") for every value but an exact\n"
     "## half, a value exactly between two n-decimal results, which goes to the even one: glibc's and musl's rule (Linux, the\n"
     "## web). The Windows build's printf rounds that half away from zero, so \"%.1f\" % -22.25 read \"-22.3\" there and \"-22.2\"\n"
     "## on Linux and the web, in text the golden hashes (the swing planner's feed line; CI, 2026-11-22). Any other value is\n"
     "## printed alike everywhere (17 significant digits name a double, so no platform's rounding lands a non-half on a half).\n"
     "## A half at n decimals is exactly k + j / 2^(n+1) with j odd, so the test scales by a power of two and is exact; so is\n"
     "## |v| * 10^n for it, below 2^52. Use it for any number a sim text shows with decimals.\n"
     "static func fixed(v: float, n: int) -> String:\n"
     "\tvar a: float = absf(v)\n"
     "\tvar p: int = 1\n"
     "\tfor i in range(n):\n"
     "\t\tp *= 10\n"
     "\tvar fr: float = (a - floorf(a)) * float(1 << (n + 1))\n"
     "\tif fr == floorf(fr) and int(fr) % 2 == 1 and a * float(p) < 4.0e15:\n"
     "\t\tvar lo: int = int(floorf(a * float(p)))\n"
     "\t\tvar k: int = lo if lo % 2 == 0 else lo + 1\n"
     "\t\tvar t: String = str(k / p)\n"
     "\t\tif n > 0:\n"
     "\t\t\tt += \".\" + str(k % p).pad_zeros(n)\n"
     "\t\treturn (\"-\" if v < 0.0 else \"\") + t\n"
     "\treturn (\"%.\" + str(n) + \"f\") % v\n"
     "\n"
     "\n"
     "## A float64 from its bit pattern, 16 hex digits with the high word first."),
])
