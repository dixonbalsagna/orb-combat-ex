#!/usr/bin/env python3
# replay_sizes_hold.py <tree>: replay-sizes re-cut for World's planet-size hold (2026-11-23), on World's GW1c stack.
#  - sim/core/replay.gd: as parked, with the size table hashed as WorldPlanetSize.hashData() (the table without `offered`), so
#    un-hiding a size moves no replay's header;
#  - sim/core/tools/parity.gd: the row, re-cut on today's main, run with WorldPlanetSize.heldOverride on (it names tiny, a held
#    size), put back after; and one more assertion: flipping a size's `offered` moves no sized replay's header.
import sys, os

T = sys.argv[1]
assert os.path.isdir(os.path.join(T, ".git")) and "scratchpad" in os.path.abspath(T).replace("\\", "/"), "refusing"


def edit(rel, pairs):
    p = os.path.join(T, rel)
    s = open(p, encoding="utf-8", newline="").read()
    for a, b in pairs:
        n = s.count(a)
        assert n == 1, "%s: anchor found %d times: %r" % (rel, n, a[:90])
        s = s.replace(a, b, 1)
    open(p, "w", encoding="utf-8", newline="").write(s)
    print("edited", rel)

     "\t\tif setup.has(\"planetSize\"):   # ... and the size table, for a setup that names a size (an unnamed one, medium, hashes as before)\n"
