#!/usr/bin/env python3
# data_compare.py <dataA> <dataB>: every JSON under two data folders compared by value (comment keys skipped, 2 == 2.0).
import json, os, sys

A, B = sys.argv[1], sys.argv[2]


def strip(v):
    if isinstance(v, dict):
        return {k: strip(x) for k, x in v.items() if not k.startswith("_")}
    if isinstance(v, list):
        return [strip(x) for x in v]
    return float(v) if isinstance(v, (int, float)) and not isinstance(v, bool) else v


def rels(root):
    out = set()
    for r, _, files in os.walk(root):
        for f in files:
            if f.endswith(".json"):
                out.add(os.path.relpath(os.path.join(r, f), root).replace(os.sep, "/"))
    return out


ra, rb = rels(A), rels(B)
diff = []
for rel in sorted(ra & rb):
    try:
        if strip(json.load(open(os.path.join(A, rel), encoding="utf-8"))) != strip(json.load(open(os.path.join(B, rel), encoding="utf-8"))):
            diff.append(rel)
    except Exception as e:
        diff.append(rel + " (" + str(e)[:40] + ")")
print("differ in value:", len(diff))
for d in diff:
    print("  ", d)
print("only in the first:", len(ra - rb))
for d in sorted(ra - rb)[:40]:
    print("  ", d)
print("only in the second:", len(rb - ra))
for d in sorted(rb - ra)[:40]:
    print("  ", d)
