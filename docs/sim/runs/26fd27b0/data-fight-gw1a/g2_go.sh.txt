#!/bin/sh
# g2_go.sh <scratchpad>: why the GW1a stack + data-fight-gw1a still misses GW1a's target golden. g2 = a copy of g1 (stack + grant).
# Goldens against the target with: (1) all of the target's data; (2) only World's biomes/water.json from it; (3) only Combat's
# combat/mixed.json + phrases.json; (4) water + combat. g1's own data is put back between runs. ONE Godot job, below normal.
set -u
powershell -NoProfile -Command "(Get-Process -Id $(cat /proc/$$/winpid)).PriorityClass = 'BelowNormal'"
G="/c/Users/<user>/AppData/Local/Programs/Godot/4.7.2/Godot_v4.7.2-stable_win64_console.exe"
NS="${1:?scratchpad}"; T="$NS/g2"; O="$NS/g2o"; TA="$NS/gw1a_ta/data"
R="/c/Users/<user>/Documents/Claude Proj/Breakers-Like"
TG="$R/docs/sim/runs/6a30b4c9/gw1a-target/golden-target.json"
rm -rf "${NS:?}/g2o" "${NS:?}/g2"; mkdir -p "$O" "$NS/g2app"; export APPDATA="$(cygpath -w "$NS/g2app")"
cp -r "$NS/g1" "$T" || exit 1
cd "$T" || exit 1
case "$(pwd -P)" in */scratchpad/g2) ;; *) echo refusing; exit 1 ;; esac
cp -r data "$O/data_g1"
cp sim/core/test/golden.json "$O/golden_main.json"
one() { name="$1"; s=$(date +%s); timeout 900 "$G" --headless --path . --script res://sim/core/tools/golden.gd > "$O/$name.txt" 2>&1; echo "$name: exit $? in $(( $(date +%s) - s )) s; $(grep -o '[0-9]* matches ([0-9]* ticks)' "$O/$name.txt")"; cmp -s sim/core/test/golden.json "$TG" && echo "  $name: REPRODUCES GW1a's target" || echo "  $name: differs"; cp sim/core/test/golden.json "$O/golden_$name.json"; cp "$O/golden_main.json" sim/core/test/golden.json; cp -r "$O/data_g1/." data/; }
cp -r "$TA/." data/; one all_target_data
cp "$TA/biomes/water.json" data/biomes/water.json; one water_only
cp "$TA/combat/mixed.json" data/combat/mixed.json; cp "$TA/combat/phrases.json" data/combat/phrases.json; one combat_only
cp "$TA/biomes/water.json" data/biomes/water.json; cp "$TA/combat/mixed.json" data/combat/mixed.json; cp "$TA/combat/phrases.json" data/combat/phrases.json; one water_and_combat
echo g2 done
